### sql语句

获取事件列表
```
SELECT * FROM "EVENT" LEFT JOIN "ORIGIN" ON EVENT.EVID = ORIGIN.EVID
```


获取事件对应的到时
```
SELECT * FROM  "ASSOC"
LEFT JOIN "ARRIVAL" ON ARRIVAL.ARID = ASSOC.ARID 
WHERE ASSOC.ORID = 14525526


SELECT * FROM IDCX."ARRIVAL" 
LEFT JOIN IDCX."ASSOC" ON IDCX.ARRIVAL.ARID = IDCX.ASSOC.ARID 
WHERE IDCX.ASSOC.ORID = 14528552
```


获取事件对应的波形列表
```
SELECT * FROM IDCX.WFTAG 
LEFT JOIN IDCX.WFDISC ON IDCX.WFTAG.WFID = IDCX.WFDISC.WFID
LEFT JOIN IDCX.ARRIVAL ON IDCX.WFTAG.TAGID = IDCX.ARRIVAL.ARID
LEFT JOIN IDCX.ASSOC ON IDCX.ARRIVAL.ARID = IDCX.ASSOC.ARID
WHERE IDCX.WFTAG.TAGNAME = 'arid'


SELECT * FROM  IDCX.ASSOC 
LEFT JOIN IDCX.ARRIVAL ON  IDCX.ARRIVAL.ARID = IDCX.ASSOC.ARID   
LEFT JOIN IDCX.WFTAG  ON IDCX.WFTAG.TAGID = IDCX.ARRIVAL.ARID
LEFT JOIN IDCX.WFDISC ON IDCX.WFTAG.WFID = IDCX.WFDISC.WFID
WHERE IDCX.WFTAG.TAGNAME = 'arid'
```


波形列表  WFTAG关联WFDISC
```
SELECT * FROM WFTAG 
LEFT JOIN WFDISC ON WFTAG.WFID = WFDISC.WFID
```


事件列表
```
SELECT LAT,LON,DEPTH,TIME,ORID,EVID,JDATE,NASS,NDEF,NDP,GRN,SRN,ETYPE,DEPDP,DTYPE,MB,MBID,MS,MSID,ML,MLID,ALGORITHM,AUTH,COMMID,LDDATE FROM origin WHERE (TIME >= 1745197200) AND (TIME <= 1745211599) ORDER BY time asc
```

到时列表
```
SELECT Arrival.*, Assoc.orid, Assoc.orid, Assoc.BELIEF, Assoc.DELTA, Assoc.phase 
FROM ARRIVAL 
LEFT JOIN ASSOC ON (Arrival.Arid =Assoc.Arid) 
WHERE Assoc.orid=14525119 
```


```
SELECT Arrival.*, Assoc.orid, Assoc.orid, Assoc.BELIEF, Assoc.DELTA, Assoc.phase 
FROM ARRIVAL 
LEFT JOIN ASSOC ON Arrival.Arid =Assoc.Arid
WHERE Assoc.orid in ( SELECT ORID FROM origin WHERE (TIME >= 1745197200) AND (TIME <= 1745211599) )


```


wfdisc
```
SELECT Wfdisc.*, s.LAT, s.LON, s.STATYPE, s.REFSTA, s.DNORTH, s.DEAST FROM WFDISC LEFT JOIN STATIC.Site s ON (Wfdisc.STA =s.STA) WHERE (WFDISC.TIME BETWEEN '1745197200' AND '1745211599') OR (WFDISC.ENDTIME BETWEEN '1745197200' AND '1745211599') ORDER BY Wfdisc.STA asc,Wfdisc.chan asc,Wfdisc.time asc
```
